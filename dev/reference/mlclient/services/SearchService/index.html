    mlclient.services.SearchService ¶

    SearchService(
    rest: RestApi, scope: SearchScope | None = None
)

    Search documents, URIs and lexicon values through the REST Search API.

    Queries are sent as JSON: structured queries as {"query": ...}, CTS queries as {"ctsquery": ...}, a string as a Search API string query. Structured, CTS and combined JSON dictionaries are accepted directly. Combined-query options override matching service defaults; per-operation scope options override both. Raw CTS JSON cannot filter lexicon operations; pass a CTS query object to those operations instead. No XQuery is evaluated, so the REST user needs only the rest-reader role; a scope naming another database also needs the xdmp-eval-in privilege.

    The service carries a SearchScope: calling it, as in ml.search(database="catalog"), returns a service whose operations all use that scope, and an operation's scope argument overrides it for one request.

    Create the service using the client's REST API.

    Parameters:

    Name Type Description Default
    rest RestApi

    REST API used to send requests; no request is made here.

    		 required
    scope SearchScope | None

    Scope applied to every operation; None searches the REST server's database without restrictions or options.

    		 None

    def aggregate(
    self,
    name: str,
    function: str,
    query: SearchInput = None,
    *,
    aggregate_path: str | None = None,
    direction: Direction | None = None,
    frequency: Frequency | None = None,
    limit: int | None = None,
    pos: PositionInput = None,
    scope: SearchScope | None = None,
    database: str | None = UNSET,
    txid: str | None = UNSET,
    timeout=UNSET,
) -> ParsedValue | None:
    """Aggregate a values or tuples definition with a built-in or native plugin.

    Sends POST with named or inline options. The definition
    must exist in the scope's options or the server's default options.

    Parameters
    ----------
    name : str
        The name of a values or tuples definition in the query options.
    function : str
        Aggregate function name. Built-ins include sum, avg, count, min,
        max, median, stddev, stddev-population, variance,
        variance-population, correlation,
        covariance and covariance-population. For a native plugin function,
        also supply aggregate_path.
    query : SearchQuery | dict | str | None, default None
        Restrict the aggregated values to documents matching a structured,
        CTS or string query; None aggregates the whole lexicon.
    aggregate_path : str | None, default None
        Native plugin library path containing the aggregate function.
        Omitted or empty selects a built-in function. The plugin must already
        be installed on MarkLogic.
    direction : {'ascending', 'descending'} | None, default None
        Lexicon order deciding which values limit and pos select.
    frequency : {'item', 'fragment'} | None, default None
        Count occurrences (item) or matching fragments (fragment).
    limit : int | None, default None
        Cap the lexicon subset before applying pos; None has no cap.
    pos : int | list[int] | tuple[int, int] | None, default None
        A one-based position or an inclusive [start, end] range of values to
        aggregate. None aggregates every value.
    scope : SearchScope | None, default None
        Overrides the fields it sets on this service's scope for this request.
    database : str | None, default unset
        Database override, taking precedence over scope; None clears it.
    txid : str | None, default unset
        Transaction override, taking precedence over scope; None clears it.
    timeout : httpx.Timeout | float | None, default unset
        A per-request HTTP timeout. Unset uses the client's configured
        timeout; None disables every HTTP timeout; a number sets all four
        components to that many seconds; an httpx.Timeout overrides them.

    Returns
    -------
    ParsedValue | None
        The first aggregate value, or None when no value is returned.
        Built-in results are converted to the function's native result type.
        Plugin values retain their JSON types without lexicon-based conversion.
        Maps retain the REST map.entry structure.

    Raises
    ------
    TypeError
        If the query has an unsupported type or needs server evaluation, pos
        is not an integer or a two-item range of integers, or scope is not a
        SearchScope.
    ValueError
        If pos is not positive or not ordered, or a query value has no
        local native form.
    WrongParametersError
        If the name is blank.
    httpx.TimeoutException
        If an HTTP connect, read, write or pool timeout expires after any
        configured retries are exhausted.
    MarkLogicError
        If MarkLogic returns an error.
    """
    params = _request_params(
        self._scope,
        query,
        pos,
        scope,
        database=database,
        txid=txid,
        lexicon=True,
    )
    resp = self._rest.values.post(
        name,
        **params,
        data_format="json",
        view="aggregate",
        aggregate=function,
        aggregate_path=aggregate_path,
        direction=direction,
        frequency=frequency,
        limit=limit,
        timeout=timeout,
    )
    return _parse_aggregate(resp, function, aggregate_path)

    def values(
    self,
    name: str,
    query: SearchInput = None,
    *,
    direction: Direction | None = None,
    frequency: Frequency | None = None,
    limit: int | None = None,
    pos: PositionInput = None,
    scope: SearchScope | None = None,
    database: str | None = UNSET,
    txid: str | None = UNSET,
    timeout=UNSET,
) -> list[ValueHit]:
    """Return values of a named lexicon definition with their frequencies.

    Sends POST with named or inline options. The values
    definition must exist in the scope's options or the server's default
    options.

    Parameters
    ----------
    name : str
        The name of a values definition in the query options.
    query : SearchQuery | dict | str | None, default None
        Restrict values to documents matching a structured, CTS or string
        query; None reads the whole lexicon.
    direction : {'ascending', 'descending'} | None, default None
        Lexicon order; None uses the options' sort order.
    frequency : {'item', 'fragment'} | None, default None
        Count occurrences (item) or matching fragments (fragment).
    limit : int | None, default None
        Cap the lexicon subset before applying pos; None has no cap.
    pos : int | list[int] | tuple[int, int] | None, default None
        A one-based position or an inclusive [start, end] range of values.
        None returns every value, as the endpoint does.
    scope : SearchScope | None, default None
        Overrides the fields it sets on this service's scope for this request.
    database : str | None, default unset
        Database override, taking precedence over scope; None clears it.
    txid : str | None, default unset
        Transaction override, taking precedence over scope; None clears it.
    timeout : httpx.Timeout | float | None, default unset
        A per-request HTTP timeout. Unset uses the client's configured
        timeout; None disables every HTTP timeout; a number sets all four
        components to that many seconds; an httpx.Timeout overrides them.

    Returns
    -------
    list[ValueHit]
        Values converted from their reported XML Schema type, in lexicon
        order, with their frequencies; [] when there are none.

    Raises
    ------
    TypeError
        If the query has an unsupported type or needs server evaluation, pos
        is not an integer or a two-item range of integers, or scope is not a
        SearchScope.
    ValueError
        If pos is not positive or not ordered, a query value has no local
        native form, or the name refers to a tuples definition.
    WrongParametersError
        If the name is blank, or direction or frequency is outside its
        supported set.
    httpx.TimeoutException
        If an HTTP connect, read, write or pool timeout expires after any
        configured retries are exhausted.
    MarkLogicError
        If MarkLogic returns an error.
    """
    params = _request_params(
        self._scope,
        query,
        pos,
        scope,
        database=database,
        txid=txid,
        lexicon=True,
    )
    resp = self._rest.values.post(
        name,
        **params,
        data_format="json",
        view="values",
        direction=direction,
        frequency=frequency,
        limit=limit,
        timeout=timeout,
    )
    return _parse_values(resp)
